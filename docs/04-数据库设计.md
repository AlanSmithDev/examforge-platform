# 数据库设计 —— 智卷云

> 版本：v1.0 ｜ MySQL 8.0，utf8mb4；分片库用 ShardingSphere 接管逻辑表
> 设计原则：题目内容结构化（JSON Block）而非富文本字符串；图形独立成实体；统计与内容分离；所有表含 `create_time/update_time/deleted`（下略）。

## 1. ER 总览

```
subject(学科) ─┬─ catalog_node(教材章节树)      knowledge_point(知识点树)
               │                                   │
textbook_version(教材版本)                      question_knowledge(题目↔知识点)
                                                   │
figure(图形) ─── question_content.blocks[].figure_id ── question(题目主表,分片)
                    │                                   │ ├─ question_option(选项)
                    │                                   │ ├─ question_answer(答案/解析)
                    │                                   ├─ question_stats(统计,分离)
                    │                                   └─ question_audit(审核流水)
user ─┬─ basket_item(试题篮)   paper_blueprint(组卷蓝图) ── paper(试卷,分片)
      ├─ paper ─ paper_section ─ paper_question       │
      ├─ favorite(收藏)                               └─ download_task(导出任务)
      ├─ practice_record ─ wrong_question(错题本)
      └─ member(会员) ─ point_account(点数) ─ order ─ payment
school ─ school_question(校本题库) ─ school_member(成员)
```

## 2. 内容域核心表（题库，分片：`subject_id` 分库、`question_id` 分表）

```sql
CREATE TABLE `question` (
  `id`            BIGINT       NOT NULL COMMENT '雪花ID',
  `subject_id`    TINYINT      NOT NULL COMMENT '1语文 2数学 3英语…',
  `question_type` TINYINT      NOT NULL COMMENT '1单选 2多选 3填空 4解答 5判断 6作图',
  `difficulty`    TINYINT      NOT NULL COMMENT '1易 2较易 3中 4较难 5难',
  `source_type`   TINYINT      NOT NULL COMMENT '1高考真题 2模拟 3名校月考 4原创 5AI生成',
  `source_desc`   VARCHAR(200) COMMENT '如：2025 新高考Ⅰ卷 T21',
  `year`          SMALLINT, `region_code` VARCHAR(10),
  `status`        TINYINT      NOT NULL DEFAULT 0 COMMENT '0草稿 1审核中 2上架 3下架 4驳回',
  `aigc_flag`     TINYINT      NOT NULL DEFAULT 0 COMMENT 'AIGC 标识',
  `use_count`     INT          DEFAULT 0 COMMENT '组卷使用次数',
  `content_version` INT        NOT NULL DEFAULT 1 COMMENT '内容版本号(缓存失效用)',
  KEY `idx_type_diff` (`question_type`,`difficulty`),
  KEY `idx_source` (`source_type`,`year`,`region_code`)
) COMMENT '题目主表(纯元数据,内容在 content 表)';

CREATE TABLE `question_content` (
  `question_id` BIGINT NOT NULL,
  `blocks`      JSON   NOT NULL COMMENT '结构化内容块数组,见下',
  `latex_text`  TEXT   COMMENT '纯LaTeX提取文本(检索冗余,便于like)',
  KEY `idx_question` (`question_id`)
) COMMENT '1:1 内容表。blocks 示例:
[{"t":"text","v":"已知函数 "},
 {"t":"latex","v":"f(x)=x^{2}-a\\ln x"},
 {"t":"figure","id":123456789},
 {"t":"text","v":"，若 f(x) 存在极值，求 a 的取值范围。"}]';

CREATE TABLE `question_option` (
  `id` BIGINT, `question_id` BIGINT, `label` CHAR(1) COMMENT 'A/B/C/D',
  `blocks` JSON COMMENT '选项内容(可含公式/图形)', `is_answer` TINYINT DEFAULT 0,
  KEY `idx_q` (`question_id`)
) COMMENT '选择题选项';

CREATE TABLE `question_answer` (
  `question_id` BIGINT,
  `answer`      JSON COMMENT '答案(选择:字母数组/填空:表达式/解答:步骤blocks)',
  `analysis`    JSON COMMENT '解析 blocks(可含多解法 solve数组)',
  `tips`        VARCHAR(500) COMMENT '点拨/易错点',
  KEY `idx_q` (`question_id`)
) COMMENT '答案与解析';

CREATE TABLE `question_knowledge` (
  `question_id` BIGINT, `kp_id` INT COMMENT '知识点ID',
  `is_primary`  TINYINT DEFAULT 0 COMMENT '主考点',
  `method_tag`  VARCHAR(50) COMMENT '解题方法标签,如:构造函数',
  KEY `idx_kp` (`kp_id`,`question_id`)
) COMMENT '题目↔知识点(ES同步源)';

CREATE TABLE `question_stats` (
  `question_id` BIGINT PRIMARY KEY,
  `compose_count` INT COMMENT '被组卷次数', `correct_rate` DECIMAL(5,2),
  `favor_count` INT, `feedback_count` INT
) COMMENT '统计分离表(高频更新,避免打主表)';

CREATE TABLE `question_audit` (
  `id` BIGINT, `question_id` BIGINT, `node` TINYINT COMMENT '1机审 2一审 3终审',
  `auditor_id` BIGINT, `result` TINYINT, `remark` VARCHAR(500)
) COMMENT '审核流水(机审含SymPy验算结果)';
```

## 3. 知识点与教材体系（不分片，全局缓存）

```sql
CREATE TABLE `subject` ( `id` TINYINT PRIMARY KEY, `name` VARCHAR(20)) COMMENT '学科';
CREATE TABLE `textbook_version` (
  `id` INT PRIMARY KEY, `subject_id` TINYINT, `name` VARCHAR(50) COMMENT '人教A版2019',
  `region` VARCHAR(20) COMMENT '主要使用地区'
) COMMENT '教材版本';
CREATE TABLE `catalog_node` (
  `id` INT, `version_id` INT, `parent_id` INT, `name` VARCHAR(100),
  `level` TINYINT COMMENT '1册 2章 3节', `sort` INT,
  KEY `idx_ver` (`version_id`,`parent_id`)
) COMMENT '教材章节树';
CREATE TABLE `knowledge_point` (
  `id` INT PRIMARY KEY, `subject_id` TINYINT, `parent_id` INT,
  `name` VARCHAR(100), `level` TINYINT COMMENT '1领域 2考点 3子考点',
  `exam_weight` TINYINT COMMENT '高考权重1-5', `sort` INT,
  KEY `idx_sub` (`subject_id`,`parent_id`)
) COMMENT '知识点树(全量缓存Caffeine)';
```

## 5. 组卷域（paper 按 `user_id` 分片）

```sql
CREATE TABLE `paper_blueprint` (
  `id` BIGINT PRIMARY KEY, `user_id` BIGINT, `subject_id` TINYINT,
  `scope` JSON COMMENT '范围:章节/知识点/年份地区',
  `structure` JSON COMMENT '题型结构:[{type,count,score}]',
  `difficulty_dist` JSON COMMENT '难度分布:{1:0.2,2:0.3,3:0.3,4:0.15,5:0.05}',
  `kp_coverage` JSON COMMENT '必覆盖知识点',
  `dup_control` JSON COMMENT '排除:本人30天已用题/同校已用题',
  `total_score` SMALLINT, `duration` SMALLINT
) COMMENT '组卷蓝图(智能/细目表/AI共用)';

CREATE TABLE `paper` (
  `id` BIGINT, `user_id` BIGINT, `subject_id` TINYINT,
  `title` VARCHAR(200), `paper_type` TINYINT COMMENT '1组卷 2真题改编 3平行卷A/B 4校本',
  `blueprint_id` BIGINT, `status` TINYINT COMMENT '0编辑 1定稿 2已导出',
  `total_score` SMALLINT, `paper_year` SMALLINT,
  KEY `idx_user` (`user_id`,`update_time`)
) COMMENT '试卷(检索走ES宽表)';

CREATE TABLE `paper_section` (
  `id` BIGINT, `paper_id` BIGINT, `name` VARCHAR(50) COMMENT '一、选择题',
  `sort` INT, `score_note` VARCHAR(50) COMMENT '每题5分,共40分'
) COMMENT '大题结构';

CREATE TABLE `paper_question` (
  `paper_id` BIGINT, `section_id` BIGINT, `question_id` BIGINT,
  `sort` INT, `score` TINYINT, `question_snapshot` JSON COMMENT '定稿时内容快照',
  KEY `idx_paper` (`paper_id`,`sort`)
) COMMENT '试卷题目(快照保证定稿后内容不漂移)';

CREATE TABLE `download_task` (
  `id` BIGINT, `paper_id` BIGINT, `user_id` BIGINT,
  `format` TINYINT COMMENT '1PDF 2Word 3答题卡',
  `options` JSON COMMENT '纸张/栏数/答案位置/字号',
  `status` TINYINT COMMENT '0排队 1渲染 2完成 3失败',
  `file_url` VARCHAR(300) COMMENT '签名URL由网关层实时签发',
  `retry_count` TINYINT DEFAULT 0, `error_msg` VARCHAR(500)
) COMMENT '导出任务(异步,MQ驱动)';
```

## 8. 分片与治理策略汇总

| 表群 | 分片键 | 策略 | 备注 |
|---|---|---|---|
| question 系列 | subject_id + question_id | 16 库×64 表 | ES 为唯一复杂查询入口 |
| paper 系列 | user_id | 16 库×32 表 | 后台运营查询走 ES 宽表 |
| practice 系列 | user_id | 16 库×64 表 | 90 天外归档 |
| 配置类（知识点/模板/学科） | 不分片 | 每库一份+全局缓存 | 永不过期缓存+MQ 刷新 |
| 交易类 | user_id | 独立库，强一致 | 本地消息表 |

- 所有分片表禁用跨片 join 与无分片键更新；分布式 ID 用时间有序雪花（利于 InnoDB 顺序写）。
- DDL 变更走工单+gh-ost 在线变更；大表（question）变更窗口避开模考季。
